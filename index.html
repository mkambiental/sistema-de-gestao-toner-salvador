copias integer; dias numeric(12,2); BEGIN IF NEW.deletado IS TRUE THEN UPDATE public.ciclos_toner SET deletado=true, data_delecao=COALESCE(NEW.data_delecao, now()) WHERE troca_fim_id=NEW.id; RETURN NEW; END IF; UPDATE public.ciclos_toner SET deletado=false, data_delecao=NULL WHERE troca_fim_id=NEW.id; IF NEW.contador_troca IS NOT NULL THEN INSERT INTO public.leituras_contador (impressora_id, cor, contador, origem_tipo, origem_id, status_origem, data_leitura, observacoes) VALUES (NEW.impressora_id, NEW.cor, NEW.contador_troca, 'TROCA', NEW.id, 'TROCA', NEW.data_troca, NEW.observacoes) ON CONFLICT (origem_tipo, origem_id) DO UPDATE SET impressora_id=EXCLUDED.impressora_id, cor=EXCLUDED.cor, contador=EXCLUDED.contador, status_origem=EXCLUDED.status_origem, data_leitura=EXCLUDED.data_leitura, observacoes=EXCLUDED.observacoes; SELECT id INTO leitura_troca_id FROM public.leituras_contador WHERE origem_tipo='TROCA' AND origem_id=NEW.id; END IF; SELECT t.id, t.data_troca, t.contador_troca, lc.id AS leitura_id INTO anterior FROM public.trocas_toner t LEFT JOIN public.leituras_contador lc ON lc.origem_tipo='TROCA' AND lc.origem_id=t.id WHERE t.impressora_id=NEW.impressora_id AND t.cor=NEW.cor AND t.deletado IS NOT TRUE AND t.id<>NEW.id AND t.data_troca <= NEW.data_troca ORDER BY t.data_troca DESC, t.id DESC LIMIT 1; IF anterior.id IS NOT NULL THEN IF NEW.contador_troca IS NOT NULL AND anterior.contador_troca IS NOT NULL AND NEW.contador_troca >= anterior.contador_troca THEN copias := NEW.contador_troca - anterior.contador_troca; ELSE copias := NULL; END IF; dias := GREATEST(0, EXTRACT(EPOCH FROM (NEW.data_troca - anterior.data_troca)) / 86400.0); SELECT id INTO ciclo_existente FROM public.ciclos_toner WHERE troca_fim_id=NEW.id LIMIT 1; IF ciclo_existente IS NOT NULL THEN UPDATE public.ciclos_toner SET impressora_id=NEW.impressora_id, cor=NEW.cor, solicitacao_recebida_id=NEW.solicitacao_id, leitura_inicio_id=anterior.leitura_id, leitura_fim_id=leitura_troca_id, contador_inicio=anterior.contador_troca, contador_fim=NEW.contador_troca, copias_produzidas=copias, data_inicio=anterior.data_troca, data_fim=NEW.data_troca, dias_duracao=dias, troca_inicio_id=anterior.id, tipo_ciclo='TROCA' WHERE id=ciclo_existente; ELSE INSERT INTO public.ciclos_toner (impressora_id, cor, solicitacao_recebida_id, leitura_inicio_id, leitura_fim_id, contador_inicio, contador_fim, copias_produzidas, data_inicio, data_fim, dias_duracao, troca_inicio_id, troca_fim_id, tipo_ciclo) VALUES (NEW.impressora_id, NEW.cor, NEW.solicitacao_id, anterior.leitura_id, leitura_troca_id, anterior.contador_troca, NEW.contador_troca, copias, anterior.data_troca, NEW.data_troca, dias, anterior.id, NEW.id, 'TROCA'); END IF; END IF; RETURN NEW; END; $$; DROP TRIGGER IF EXISTS trg_sgt_troca_toner ON public.trocas_toner; CREATE TRIGGER trg_sgt_troca_toner AFTER INSERT OR UPDATE OF impressora_id, cor, data_troca, contador_troca, solicitacao_id, observacoes, deletado ON public.trocas_toner FOR EACH ROW EXECUTE FUNCTION public.sgt_registrar_troca_toner();
